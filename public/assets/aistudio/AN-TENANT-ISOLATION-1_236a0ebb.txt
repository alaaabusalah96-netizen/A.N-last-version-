export async function withTenantTransaction<T>(
  executorOrPool: /* use existing access to pg Pool if exposed, or extend adapter */,
  companyId: string,
  fn: (clientQuery: { query: (sql: string, params?: unknown[]) => Promise<...> }) => Promise<T>
): Promise<T>

Rules:
- if !companyId.trim() → throw TENANT_CONTEXT_REQUIRED
- checkout one client from pool
- BEGIN
- SELECT set_config('app.current_tenant_id', $1, true)
- run fn with client-bound query only
- COMMIT / ROLLBACK
- release in finally
- never session-level SET without transaction local (third arg true)

If current DbExecutor hides Pool: extend PostgresqlAdapter with
getPool() for internal use OR add withTenantTransaction on adapter.
Do not invent a second parallel DB framework.

Wire tenantContext.run(companyId, ...) inside the transaction if useful.

database/migrations/006_tenant_demo_and_rls.sql

-- Demo table for isolation proof
CREATE TABLE IF NOT EXISTS tenant_demo_items (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id UUID NOT NULL REFERENCES companies(id) ON DELETE CASCADE,
  label TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_tenant_demo_company ON tenant_demo_items(company_id);

ALTER TABLE tenant_demo_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE tenant_demo_items FORCE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS tenant_demo_isolation ON tenant_demo_items;
CREATE POLICY tenant_demo_isolation ON tenant_demo_items
  FOR ALL
  USING (company_id::text = NULLIF(current_setting('app.current_tenant_id', true), ''))
  WITH CHECK (company_id::text = NULLIF(current_setting('app.current_tenant_id', true), ''));

-- company_memberships: enable RLS so reads/writes respect tenant when context set
ALTER TABLE company_memberships ENABLE ROW LEVEL SECURITY;
-- Prefer FORCE only if app role is not table owner; document role strategy
CREATE POLICY memberships_tenant_select ON company_memberships
  FOR SELECT
  USING (
    company_id::text = NULLIF(current_setting('app.current_tenant_id', true), '')
    OR current_setting('app.current_tenant_id', true) = ''
  );
-- Note: empty setting may mean "policy allows all" — better:
-- For app role: policy USING (company_id::text = current_setting(...))
-- and platform operations use a bypass role OR set a special platform context.

-- Minimum acceptable for this package: demo table RLS works under withTenantTransaction; memberships filtered by company_id in SQL always.
