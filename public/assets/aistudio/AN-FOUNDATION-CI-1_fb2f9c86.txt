Create scripts/runFoundationBattery.ts:

- exit code 0 if only Pass+Blocked (or exit 1 when REQUIRE_DB=1 and any Blocked — document choice: prefer exit 1 on Blocked when CI has DB so Blocked is unexpected)

2) .github/workflows/foundation-ci.yml
──────────────────────────────────────────────────────────────────
on: push/PR to main
jobs:
  foundation:
    runs-on: ubuntu-latest
    services:
      postgres:
        image: postgres:16
        env:
          POSTGRES_USER: an_test
          POSTGRES_PASSWORD: an_test_pass
          POSTGRES_DB: an_foundation
        ports: ["5432:5432"]
        options: health-cmd pg_isready ...
    steps:
      - checkout
      - setup-node 22 + cache npm
      - npm ci
      - npm run typecheck
      - npm run build
      - npm run migrate  (if script exists)
      - npm run test:foundation
        env:
          DATABASE_URL: postgresql://an_test:an_test_pass@localhost:5432/an_foundation
          NODE_ENV: test
          REQUIRE_DB: "1"
          JWT_SECRET or SESSION secrets if required by tests: use CI dummy values

Do not commit secrets. Document PLATFORM_ADMIN_* only if bootstrap tests need them — use random CI-only values in env block.
