-- Package: AN-FOUNDATION-SECURITY-BASELINE-1
-- 
-- 1. Enforces Row Level Security (RLS) across all application tables.

BEGIN

  END IF;
END $$;

  WITH CHECK (company_id = NULLIF(current_setting('app.current_tenant_id', true), '')::uuid);

-- 4. Enforce strict table privileges on audit_logs and schema_migrations

REVOKE ALL ON TABLE schema_migrations FROM an_app;
GRANT SELECT ON TABLE schema_migrations TO an_app;

-- 5. Revoke unnecessary application-table privileges from anon and authenticated
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'anon') THEN
    REVOKE ALL ON ALL TABLES IN SCHEMA public FROM anon;
    REVOKE ALL ON ALL SEQUENCES IN SCHEMA public FROM anon;
    REVOKE ALL ON ALL ROUTINES IN SCHEMA public FROM anon;
    REVOKE CREATE ON SCHEMA public FROM anon;
  END IF;

  IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'authenticated') THEN
    REVOKE ALL ON ALL TABLES IN SCHEMA public FROM authenticated;
    REVOKE ALL ON ALL SEQUENCES IN SCHEMA public FROM authenticated;
    REVOKE ALL ON ALL ROUTINES IN SCHEMA public FROM authenticated;
    REVOKE CREATE ON SCHEMA public FROM authenticated;
  END IF;
END $$;

-- 6. Configure default privileges for objects created by migration owner
DO $$
BEGIN
  -- Prevent future application tables, sequences, and routines from exposing to anon & authenticated
  IF EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'anon') AND EXISTS (SELECT 1 FROM pg_roles WHERE rolname = 'authenticated') THEN
    EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public REVOKE ALL ON TABLES FROM anon, authenticated;', CURRENT_USER);
    EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public REVOKE ALL ON SEQUENCES FROM anon, authenticated;', CURRENT_USER);
    EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public REVOKE ALL ON ROUTINES FROM anon, authenticated;', CURRENT_USER);
  END IF;

  -- Revoke global PUBLIC execution default on routines in schema public
  EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public REVOKE EXECUTE ON ROUTINES FROM PUBLIC;', CURRENT_USER);

  -- Ensure future tables and sequences created by migration owner grant required least-privilege to an_app
  EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public GRANT SELECT, INSERT, UPDATE, DELETE ON TABLES TO an_app;', CURRENT_USER);
  EXECUTE format('ALTER DEFAULT PRIVILEGES FOR ROLE %I IN SCHEMA public GRANT USAGE, SELECT ON SEQUENCES TO an_app;', CURRENT_USER);
EXCEPTION
  WHEN undefined_object THEN
    NULL;
END $$;

import { DbExecutor } from './types.js';

export interface DataApiGuardOptions {
  requireRoles?: boolean;
}

function sanitizeError(err: unknown): string {
  const msg = err instanceof Error ? err.message : String(err);
  return msg
    .replace(/password=[^\s]+/gi, 'password=[REDACTED]')
    .replace(/postgres(?:ql)?:\/\/[^@\s]+@[^\s]+/gi, 'postgresql://[REDACTED_AUTH]@[HOST]');
}

export async function assertNoDataApiExposure(
  executor: DbExecutor,
  options: DataApiGuardOptions = {}
): Promise<string[]> {
  if (!executor || typeof executor.query !== 'function') {
    throw new Error('[dataApiGuard] FATAL: Invalid executor provided — check failed closed.');
  }

  const requireRoles = options.requireRoles ?? true;
  const untrustedRoles = ['anon', 'authenticated'];
  const violations: string[] = [];

  try {
    for (const role of untrustedRoles) {
      // 1. Role existence check
      const roleCheck = await executor.query<{ exists: boolean }>(
        'SELECT to_regrole($1) IS NOT NULL as exists;',
        [role]
      );

      const exists = Boolean(roleCheck.rows[0]?.exists);
      if (!exists) {
        if (requireRoles) {
          throw new Error(`[dataApiGuard] FATAL: Required Data API role '${role}' is missing.`);
        }
        continue;
      }

      // 2. Schema CREATE privilege check
      const schemaCheck = await executor.query<{ has_create: boolean }>(
        "SELECT has_schema_privilege($1, 'public', 'CREATE') as has_create;",
        [role]
      );
      if (schemaCheck.rows[0]?.has_create) {
        violations.push(`Role '${role}' has unauthorized CREATE privilege on schema public.`);
      }

      // 3. Public tables and views privilege check (SELECT, INSERT, UPDATE, DELETE, TRUNCATE, REFERENCES, TRIGGER)
      const tablePrivs = await executor.query<{ tablename: string; priv: string }>(
        `SELECT c.relname as tablename, p.priv
         FROM pg_class c
         JOIN pg_namespace n ON n.oid = c.relnamespace
         CROSS JOIN (VALUES ('SELECT'), ('INSERT'), ('UPDATE'), ('DELETE'), ('TRUNCATE'), ('REFERENCES'), ('TRIGGER')) as p(priv)
         WHERE n.nspname = 'public'
           AND c.relkind IN ('r', 'v', 'm', 'p')
           AND has_table_privilege($1, c.oid, p.priv);`,
        [role]
      );

      for (const row of tablePrivs.rows) {
        violations.push(`Role '${role}' has unauthorized ${row.priv} privilege on table/view '${row.tablename}'.`);
      }

      // 4. Public sequences privilege check (USAGE, SELECT, UPDATE)
      const seqPrivs = await executor.query<{ seqname: string; priv: string }>(
        `SELECT c.relname as seqname, p.priv
         FROM pg_class c
         JOIN pg_namespace n ON n.oid = c.relnamespace
         CROSS JOIN (VALUES ('USAGE'), ('SELECT'), ('UPDATE')) as p(priv)
         WHERE n.nspname = 'public'
           AND c.relkind = 'S'
           AND has_sequence_privilege($1, c.oid, p.priv);`,
        [role]
      );

      for (const row of seqPrivs.rows) {
        violations.push(`Role '${role}' has unauthorized ${row.priv} privilege on sequence '${row.seqname}'.`);
      }

      // 5. Public callable functions privilege check (EXECUTE)
      // Exclude trigger-returning functions from callable RPC check
      const funcPrivs = await executor.query<{ proname: string }>(
        `SELECT p.proname
         FROM pg_proc p
         JOIN pg_namespace n ON n.oid = p.pronamespace
         WHERE n.nspname = 'public'
           AND p.prorettype NOT IN ('trigger'::regtype, 'event_trigger'::regtype)
           AND has_function_privilege($1, p.oid, 'EXECUTE');`,
        [role]
      );

      for (const row of funcPrivs.rows) {
        violations.push(`Role '${role}' has unauthorized EXECUTE privilege on callable function '${row.proname}'.`);
      }
    }

    return violations;
  } catch (err: unknown) {
    if (err instanceof Error && err.message.startsWith('[dataApiGuard]')) {
      throw err;
    }
    throw new Error(`[dataApiGuard] FATAL: Data API guard query failed (fail-closed): ${sanitizeError(err)}`);
  }
}

        rows: rows as T[],
        rowCount: rows.length,

import fs from 'fs';
import path from 'path';
import { check, blocked, summary } from './lib/testkit.js';

console.log('--- Running RLS & Table Ownership Coverage Verification (PHASE 4 / RLS BASELINE) ---');

const rootDir = process.cwd();

interface TableOwnership {
  table: string;
  scope: 'global' | 'shared' | 'tenant';
  reason: string;
  allowOptionalCompanyId?: boolean;
}

async function runTests() {
  const registryPath = path.join(rootDir, 'database/table_ownership.json');
  await check('database/table_ownership.json exists', fs.existsSync(registryPath));

  const registry: TableOwnership[] = JSON.parse(fs.readFileSync(registryPath, 'utf8'));
  const registryMap = new Map<string, TableOwnership>(registry.map((r) => [r.table, r]));

  const dbUrl = process.env.DATABASE_URL;
  if (!dbUrl || dbUrl.trim() === '') {
    blocked('DATABASE_URL is not set. Skipping live DB table ownership & RLS queries.');
    const { exitCode } = summary();
    process.exit(exitCode);
    return;
  }

  try {
    const { getDefaultExecutor } = await import('../src/server/db/index.js');
    const executor = await getDefaultExecutor();
    if (!executor) {
      blocked('Unable to connect to DB for table ownership verification.');
      const { exitCode } = summary();
      process.exit(exitCode);
      return;
    }

    // 1. Fetch all tables in public schema
    const tablesSql = `
      SELECT c.oid, c.relname as table_name, c.relrowsecurity, c.relforcerowsecurity
      FROM pg_class c
      JOIN pg_namespace n ON n.oid = c.relnamespace
      WHERE n.nspname = 'public' AND c.relkind = 'r';
    `;
    const tablesRes = await executor.query<{ oid: number; table_name: string; relrowsecurity: boolean; relforcerowsecurity: boolean }>(tablesSql);
    const dbTablesMap = new Map(tablesRes.rows.map((r) => [r.table_name, r]));

    // Rule 1: Every table in DB MUST be registered in table_ownership.json (fail on unknown tables)
    for (const tableName of dbTablesMap.keys()) {
      await check(`Table '${tableName}' is registered in table_ownership.json`, registryMap.has(tableName));
    }

    // Rule 2: Every table in table_ownership.json MUST exist in DB
    for (const regItem of registry) {
      await check(`Registered table '${regItem.table}' exists in database`, dbTablesMap.has(regItem.table));
      await check(`Registered table '${regItem.table}' has non-empty reason`, Boolean(regItem.reason && regItem.reason.trim()));

      const dbTable = dbTablesMap.get(regItem.table);
      if (!dbTable) continue;

      // Rule 3: EVERY table MUST have RLS enabled (relrowsecurity = true)
      await check(
        `Table '${regItem.table}' has RLS enabled (relrowsecurity=true)`,
        dbTable.relrowsecurity === true
      );

      // Fetch columns for this table
      const colsSql = `
        SELECT column_name, is_nullable
        FROM information_schema.columns
        WHERE table_schema = 'public' AND table_name = $1;
      `;
      const colsRes = await executor.query<{ column_name: string; is_nullable: string }>(colsSql, [regItem.table]);
      const colNames = new Set(colsRes.rows.map((c) => c.column_name));
      const companyIdCol = colsRes.rows.find((c) => c.column_name === 'company_id');

      // Fetch policies for this table
      const polSql = `
        SELECT p.polname,
               p.polcmd,
               pg_get_expr(p.polqual, p.polrelid) as qual,
               pg_get_expr(p.polwithcheck, p.polrelid) as with_check,
               ARRAY(
                 SELECT r.rolname FROM pg_roles r WHERE r.oid = ANY(p.polroles)
               ) as roles
        FROM pg_policy p
        WHERE p.polrelid = $1;
      `;
      const polRes = await executor.query<{ polname: string; polcmd: string; qual: string | null; with_check: string | null; roles: string[] }>(
        polSql,
        [dbTable.oid]
      );

      if (regItem.scope === 'global' || regItem.scope === 'shared') {
        if (regItem.scope === 'global') {
          await check(`Global table '${regItem.table}' does NOT contain company_id column`, !colNames.has('company_id'));
        }

        // Must have an explicit policy targeting an_app
        const hasAppPolicy = polRes.rows.some((p) => p.roles.includes('an_app'));
        await check(
          `${regItem.scope.toUpperCase()} table '${regItem.table}' has explicit policy targeting 'an_app'`,
          hasAppPolicy
        );
      } else if (regItem.scope === 'tenant') {
        // Tenant scope requirements
        await check(`Tenant table '${regItem.table}' contains company_id column`, Boolean(companyIdCol));
        await check(`Tenant table '${regItem.table}' company_id column is NOT NULL`, companyIdCol?.is_nullable === 'NO');

        // Robust catalog Foreign Key Check
        const fkCatalogSql = `
          SELECT 
            con.conname AS constraint_name,
            c_src.relname AS source_table,
            a_src.attname AS source_column,
            c_ref.relname AS ref_table,
            a_ref.attname AS ref_column,
            con.convalidated AS is_validated
          FROM pg_constraint con
          JOIN pg_class c_src ON c_src.oid = con.conrelid
          JOIN pg_namespace n_src ON n_src.oid = c_src.relnamespace
          JOIN pg_class c_ref ON c_ref.oid = con.confrelid
          JOIN pg_namespace n_ref ON n_ref.oid = c_ref.relnamespace
          JOIN pg_attribute a_src ON a_src.attrelid = con.conrelid AND a_src.attnum = ANY(con.conkey)
          JOIN pg_attribute a_ref ON a_ref.attrelid = con.confrelid AND a_ref.attnum = ANY(con.confkey)
          WHERE con.contype = 'f'
            AND n_src.nspname = 'public'
            AND n_ref.nspname = 'public'
            AND c_src.relname = $1
            AND a_src.attname = 'company_id'
            AND c_ref.relname = 'companies'
            AND a_ref.attname = 'id'
            AND con.convalidated = true;
        `;
        const fkRes = await executor.query(fkCatalogSql, [regItem.table]);
        await check(
          `Tenant table '${regItem.table}' has valid active FK on company_id referencing companies(id)`,

function redactSensitive(text: string): string {

import { defaultUserStore, User, UserStore } from './userStore.js';
import { defaultSessionStore, Session, SessionStore } from './sessionStore.js';
import { InvalidCredentialsError, ValidationError } from '../http/errors.js';

export interface LoginResult {
  session: Session;
  user: Omit<User, 'passwordHash'>;
}

// Pre-computed dummy Argon2 hash for constant-time comparison when username is missing
let DUMMY_PASSWORD_HASH: string | null = null;
async function getDummyHash(): Promise<string> {
  if (!DUMMY_PASSWORD_HASH) {
    DUMMY_PASSWORD_HASH = await hashPassword('dummy_constant_time_comparison_password');
  }
  return DUMMY_PASSWORD_HASH;
}

export class AuthService {
  constructor(
    private userStore: UserStore = defaultUserStore,
    private sessionStore: SessionStore = defaultSessionStore
  ) {}

  async login(username: string, plainPassword: string): Promise<LoginResult> {
    if (!username || typeof username !== 'string' || !plainPassword || typeof plainPassword !== 'string') {
      throw new ValidationError('Username and password are required');
    }

    const normalizedUsername = username.trim().toLowerCase();

    if (normalizedUsername.length > 150 || plainPassword.length < 8 || plainPassword.length > 1024) {
      throw new InvalidCredentialsError();
    }

    const user = await this.userStore.findByUsername(normalizedUsername);

    if (!user) {
      // Execute constant-time dummy password verification to prevent username enumeration via timing
      const dummyHash = await getDummyHash();
      await verifyPassword(plainPassword, dummyHash);
      throw new InvalidCredentialsError();
    }

    const isValid = await verifyPassword(plainPassword, user.passwordHash);
    if (!isValid) {
      throw new InvalidCredentialsError();
    }

    const session = await this.sessionStore.createSession(user.id);
    const { passwordHash: _, ...safeUser } = user;

    return { session, user: safeUser };
  }

  async logout(sessionId: string): Promise<boolean> {
    if (!sessionId) return false;
    return await this.sessionStore.revokeSession(sessionId);
  }

  async revokeAllSessionsForUser(userId: string): Promise<number> {
    if (!userId) return 0;
    return await this.sessionStore.revokeAllForUser(userId);
  }

  async getSessionUser(sessionId: string): Promise<Omit<User, 'passwordHash'> | null> {
    const session = await this.sessionStore.getSession(sessionId);
    if (!session) return null;

    const user = await this.userStore.findById(session.userId);
    if (!user) return null;

    const { passwordHash: _, ...safeUser } = user;
    return safeUser;
  }

  /**
   * Initializes initial platform administrator account securely from environment variables.
   * Requires password length >= 12 characters.
   */
  async bootstrapPlatformAdminFromEnv(): Promise<boolean> {
    const rawUsername = process.env.PLATFORM_ADMIN_USERNAME;
    const rawPassword = process.env.PLATFORM_ADMIN_PASSWORD;

    const hasUsername = Boolean(rawUsername && rawUsername.trim() !== '');
    const hasPassword = Boolean(rawPassword && rawPassword.trim() !== '');

    // Requirement 4: If only one administrator credential is configured, fail clearly
    if ((hasUsername && !hasPassword) || (!hasUsername && hasPassword)) {
      throw new Error('[auth] FATAL: Both PLATFORM_ADMIN_USERNAME and PLATFORM_ADMIN_PASSWORD must be configured together for administrator bootstrap');
    }

    if (!hasUsername || !hasPassword) {
      return false; // Neither configured; bootstrap not requested
    }

    const username = rawUsername!.trim().toLowerCase();
    const password = rawPassword!;

    // Requirement 5: Minimum 12-character administrator password requirement
    if (password.length < 12) {
      throw new Error('[auth] FATAL: PLATFORM_ADMIN_PASSWORD must be at least 12 characters long');
    }

    try {
      const existing = await this.userStore.findByUsername(username);
      if (existing) {
        // Requirement 7 & 8: If the username already exists with a different role, report an explicit conflict
        if (existing.role !== 'platform_admin') {
          throw new Error(`[auth] FATAL: Bootstrap conflict — user '${username}' already exists with role '${existing.role}', not 'platform_admin'`);
        }
        return true; // Already exists as platform_admin
      }

      const passwordHash = await hashPassword(password);
      await this.userStore.createUser({
        username,
        passwordHash,
        role: 'platform_admin',
      });

      console.log(`[auth] Platform admin '${username}' initialized from environment variables.`);
      return true;
    } catch (err: unknown) {
      if (err instanceof Error && err.message.startsWith('[auth] FATAL:')) {
        throw err;
      }
      const sanitizedMsg = err instanceof Error ? err.message : String(err);
      throw new Error(`[auth] FATAL: Failed to bootstrap platform admin '${username}': ${sanitizedMsg}`);
    }
  }
}

export const defaultAuthService = new AuthService();

function getCookie(name: string): string | null {
  if (typeof document === 'undefined') return null;
  const value = `; ${document.cookie}`;
  const parts = value.split(`; ${name}=`);
  if (parts.length === 2) return parts.pop()?.split(';').shift() || null;
  return null;
}

export async function refreshCsrfToken(): Promise<string | null> {
  try {
    const res = await fetch('/api/auth/csrf', { credentials: 'include' });
    if (res.ok) {
      const contentType = res.headers.get('content-type') || '';
      if (contentType.includes('application/json')) {
        const data = await res.json();
        return data.csrfToken || getCookie('an_csrf');
      }
    }
  } catch (err) {
    console.error('[apiClient] Failed to refresh CSRF token:', err);
  }
  return null;
}

export async function apiFetch(path: string, init: RequestInit = {}): Promise<Response> {
  const method = (init.method || 'GET').toUpperCase();
  const isMutating = ['POST', 'PUT', 'PATCH', 'DELETE'].includes(method);

  const headers = new Headers(init.headers || {});
  headers.set('Content-Type', headers.get('Content-Type') || 'application/json');

  if (isMutating) {
    let csrfToken = getCookie('an_csrf');
    if (!csrfToken) {
      csrfToken = await refreshCsrfToken();
    }
    if (csrfToken) {
      headers.set('x-csrf-token', csrfToken);
    }
  }

  const reqInit: RequestInit = {
    ...init,
    method,
    headers,
    credentials: 'include',
  };

  let res = await fetch(path, reqInit);

  // If CSRF token invalid, refresh token and retry ONCE
  if (res.status === 403 && isMutating) {
    const clone = res.clone();
    try {
      const contentType = clone.headers.get('content-type') || '';
      if (contentType.includes('application/json')) {
        const errorData = await clone.json();
        if (errorData.code === 'CSRF_INVALID') {
          console.warn('[apiClient] CSRF token invalid. Refreshing token and retrying request...');
          const newCsrfToken = await refreshCsrfToken();
          if (newCsrfToken) {
            headers.set('x-csrf-token', newCsrfToken);
            res = await fetch(path, { ...reqInit, headers });
          }
        }
      }
    } catch {
      // Ignore JSON parse error on non-JSON response
    }
  }

  return res;
}

# Supabase / Remote PostgreSQL Connection Runbook

## 1. Safety & Production Guardrails
- Always test connection, migrations, and role provisioning on a separate disposable/staging database before touching production.
- **Never run foundation test batteries against live production databases.** Test batteries generate disposable tenants, test audit entries, and benchmark loads.
- Disabling SSL or setting `DATABASE_SSL_REJECT_UNAUTHORIZED=false` is strictly forbidden and rejected in production.
- **Never commit production credentials or secrets to source control.**

## 2. Roles & Permissions Separation
The architecture strictly enforces two separate database connections:
1. **`DATABASE_URL` (Runtime Application Connection)**:
   - Must connect exclusively using the least-privilege `an_app` role.
   - Used by the Express server, services, and tenant resolvers at runtime.
   - `an_app` possesses NOINHERIT, NOSUPERUSER, NOCREATEDB, NOCREATEROLE, NOREPLICATION, and NOBYPASSRLS.
   - Transaction poolers (`pgBouncer` / Supavisor on port 6543 or 5432) are permitted.
2. **`MIGRATION_DATABASE_URL` (Privileged Migration Connection)**:
   - Used exclusively for schema migrations (`npm run migrate`) and role provisioning (`npm run db:provision-app-role`).
   - Must connect as the database owner / migration user (e.g. `postgres`).
   - **`MIGRATION_DATABASE_URL` must never be used as a runtime fallback.**
3. **`APP_DB_PASSWORD`**:
   - Provisioning input only. Defines the password for the `an_app` role.
   - Must be at least 16 printable ASCII characters.

## 3. Platform Admin Bootstrap
- **`PLATFORM_ADMIN_USERNAME`** and **`PLATFORM_ADMIN_PASSWORD`**:
  - Used for the one-time initial platform administrator account bootstrap.
  - Both variables must be provided together; configuring only one fails startup immediately.
  - Password must be at least 12 characters.
  - Never logged in plaintext or exposed in error messages.
  - If bootstrap fails (due to database unavailability or conflict), the server halts with `process.exit(1)` and refuses to listen on HTTP.
  - Ordinary users are never automatically promoted to `platform_admin`.

## 4. Data API Roles & Security Boundary
- **`anon` & `authenticated` (Untrusted Data API Roles)**:
  - Untrusted client roles used by Supabase PostgREST / Data APIs.
  - All application tables, sequences, and callable functions are completely revoked from these roles.
  - Schema `public` `CREATE` privilege is revoked.
  - The runtime Data API guard (`dataApiGuard`) fails closed if any table, sequence, or callable function privilege is exposed to `anon` or `authenticated`.
- **`service_role` (Trusted Server Role)**:
  - A secret-only administrative Supabase server role (bypasses RLS).
  - Must never be exposed to frontend code or browser clients.
  - Kept distinct and excluded from untrusted public Data API role checks.

## 5. RLS Policy Requirements by Table Scope
As classified in `database/table_ownership.json`:
1. **Global Tables (`platform_users`, `companies`, `user_sessions`, `login_rate_events`, `schema_migrations`)**:
   - RLS enabled on all tables.
   - Must not contain `company_id`.
   - Explicit RLS policy targeting role `an_app`.
   - `schema_migrations` must remain strictly SELECT-only for `an_app`.
2. **Shared Tables (`company_memberships`, `audit_logs`)**:
   - RLS enabled on all tables.
   - Explicit RLS policy targeting role `an_app`.
   - `audit_logs` is append-only: granted SELECT and INSERT only to `an_app` (UPDATE, DELETE, TRUNCATE forbidden).
3. **Tenant Tables (`tenant_demo_items`, and all future tenant tables)**:
   - `company_id UUID NOT NULL` with active foreign key referencing `companies(id)`.
   - RLS enabled and **`FORCE ROW LEVEL SECURITY`** enabled.
   - Explicit policy targeting `an_app` enforcing `app.current_tenant_id` in both `USING` and `WITH CHECK`.
   - Permissive global policies (`USING true`) are strictly forbidden on tenant tables.

## 6. Registering New Tenant Tables
1. Add the table definition with `company_id UUID NOT NULL REFERENCES companies(id)`.
2. Enable and force RLS:
   ```sql
   ALTER TABLE <new_table> ENABLE ROW LEVEL SECURITY;
   ALTER TABLE <new_table> FORCE ROW LEVEL SECURITY;
   ```
3. Add isolated tenant policy for `an_app`:
   ```sql
   CREATE POLICY <new_table>_tenant_isolation ON <new_table>
     TO an_app
     USING (company_id = NULLIF(current_setting('app.current_tenant_id', true), '')::uuid)
     WITH CHECK (company_id = NULLIF(current_setting('app.current_tenant_id', true), '')::uuid);
   ```
